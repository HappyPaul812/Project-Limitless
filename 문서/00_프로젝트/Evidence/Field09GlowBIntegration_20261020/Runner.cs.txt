        static IEnumerator Walk(Vector2 target,bool pad){

            var q=new Queue<Vector2Int>();var prev=new Dictionary<Vector2Int,Vector2Int>();q.Enqueue(start);prev[start]=start;
            bool found=false;while(q.Count>0){var n=q.Dequeue();if(Vector2.Distance((Vector2)n*step,target-Vector2.up*.7f)<=1.15f&&Clear((Vector2)n*step,radius)){goal=n;found=true;break;}foreach(var dir in new[]{Vector2Int.up,Vector2Int.down,Vector2Int.left,Vector2Int.right}){var next=n+dir;var point=(Vector2)next*step;if(Mathf.Abs(point.x)>12||Mathf.Abs(point.y)>8||prev.ContainsKey(next)||!Clear(point,radius))continue;prev[next]=n;q.Enqueue(next);}}
            Check(found,"Route.Exists."+target+".Pad"+pad);var points=new List<Vector2>();var node=goal;while(node!=start){points.Add((Vector2)node*step);node=prev[node];}points.Reverse();
            foreach(var point in points){float end=Time.realtimeSinceStartup+3;while(Vector2.Distance(Player.transform.position,point)>.11f&&Time.realtimeSinceStartup<end){Vector2 diff=point-(Vector2)Player.transform.position;Vector2 axis=Mathf.Abs(diff.x)>Mathf.Abs(diff.y)?new Vector2(Mathf.Sign(diff.x),0):new Vector2(0,Mathf.Sign(diff.y));if(pad)InputSystem.QueueStateEvent(gamepad,new GamepadState{leftStick=axis});else InputSystem.QueueStateEvent(keyboard,new KeyboardState(axis.x>0?Key.D:axis.x<0?Key.A:axis.y>0?Key.W:Key.S));yield return Wait(.02f);}InputSystem.QueueStateEvent(keyboard,new KeyboardState());InputSystem.QueueStateEvent(gamepad,new GamepadState());yield return Wait(.03f);if(Vector2.Distance(Player.transform.position,point)>=.23f&&routeRetries++<5){File.AppendAllText(Path.Combine(Root,"RouteReplans.txt"),"blocked target="+point+" player="+Player.transform.position+" modal="+WorldModalState.IsOpen+"\n");routeAvoid.Add(new Vector2Int(Mathf.RoundToInt(point.x/.25f),Mathf.RoundToInt(point.y/.25f)));yield return Walk(target,pad);yield break;}Check(Vector2.Distance(Player.transform.position,point)<.23f,"Route.PhysicsInput."+point+".Pad"+pad);}
        }
