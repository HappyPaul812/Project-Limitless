using System;using System.IO;using System.Linq;using System.Collections;using System.Collections.Generic;using UnityEditor;using UnityEditor.SceneManagement;using UnityEngine;using UnityEngine.SceneManagement;using ProjectLimitless.Core;using ProjectLimitless.Player;using ProjectLimitless.Monster;
namespace ProjectLimitless.EditorTools {
[InitializeOnLoad] public static class EODWorldSpriteProbe {
 const string Key="Limitless.EOD.WorldSpriteProbe";static IEnumerator routine;static List<string> rows=new List<string>();static string Root=>Path.GetFullPath(Path.Combine(Application.dataPath,"../WorldSpriteProbeResults"));
 static void Isolate(){Directory.CreateDirectory(Root);GameSaveService.AuditSaveDirectory=Path.Combine(Root,"Saves");UserSettingsService.BeginAudit(Path.Combine(Root,"Settings"));UserSettingsService.SetMuteAll(true);AudioListener.volume=0;}
 static EODWorldSpriteProbe(){EditorApplication.playModeStateChanged+=s=>{if(s!=PlayModeStateChange.EnteredPlayMode||!SessionState.GetBool(Key,false))return;Isolate();routine=Run();EditorApplication.update+=Tick;};}
 public static void Launch(){if(!Application.isBatchMode||!Application.dataPath.Replace('\\','/').EndsWith("/Temp/EODIntegration20261013/Client/Assets"))throw new Exception("Isolation");Isolate();EditorSceneManager.NewScene(NewSceneSetup.EmptyScene,NewSceneMode.Single);SessionState.SetBool(Key,true);EditorApplication.EnterPlaymode();}
 static IEnumerator Run(){GameSessionData.Reset();GameSaveService.SelectSlot(1);GameSessionData.ConfigurePlayer(PlayerVisualType.Male,"환경 참조 조사");GameSessionData.SelectJob("fighter");GameSessionData.ConfigureProgress(30,0);PlayerController.SetMovementLocked(typeof(EODWorldSpriteProbe),true);
 foreach(string scene in new[]{"World_StarterVillage","Arbel","Field_05_WesternOutskirts","Field_06_ScorchedTrail","Field_07_AshenReach","Field_08_RedRift"}){MonsterEncounterService.SuppressForSeconds(3600);SceneManager.LoadScene(scene);float end=Time.realtimeSinceStartup+.7f;while(Time.realtimeSinceStartup<end)yield return null;MonsterEncounterService.SuppressForSeconds(3600);var player=UnityEngine.Object.FindAnyObjectByType<PlayerController>();if(player!=null)player.enabled=false;var sr=UnityEngine.Object.FindObjectsByType<SpriteRenderer>(FindObjectsSortMode.None).Where(x=>x.gameObject.scene.name==scene).ToArray();var missing=sr.Where(x=>x.sprite==null).ToArray();rows.Add(scene+"|RuntimeSpriteRenderers="+sr.Length+"|Missing="+missing.Length);File.WriteAllLines(Path.Combine(Root,"counts.txt"),rows);File.WriteAllLines(Path.Combine(Root,scene+"_missing.txt"),missing.Select(x=>x.name));Capture(scene);}
 File.WriteAllText(Path.Combine(Root,"status.txt"),"INVESTIGATION_COMPLETE; no asset edits");}
 static void Capture(string name){var c=Camera.main;if(c==null)return;var rt=RenderTexture.GetTemporary(1680,1200,24);var old=c.targetTexture;var pos=c.transform.position;var size=c.orthographicSize;var aspect=c.aspect;var active=RenderTexture.active;try{c.transform.position=new Vector3(0,0,-10);c.orthographicSize=7.5f;c.aspect=1.4f;c.targetTexture=rt;c.Render();RenderTexture.active=rt;var t=new Texture2D(1680,1200,TextureFormat.RGBA32,false);t.ReadPixels(new Rect(0,0,1680,1200),0,0);t.Apply();File.WriteAllBytes(Path.Combine(Root,name+".png"),t.EncodeToPNG());UnityEngine.Object.DestroyImmediate(t);}finally{c.targetTexture=old;c.transform.position=pos;c.orthographicSize=size;c.aspect=aspect;RenderTexture.active=active;RenderTexture.ReleaseTemporary(rt);}}
 static void Tick(){try{if(routine.MoveNext())return;SessionState.SetBool(Key,false);EditorApplication.Exit(0);}catch(Exception e){File.WriteAllText(Path.Combine(Root,"failure.txt"),e.ToString());EditorApplication.Exit(1);}}
}}
