    [InitializeOnLoad]
    public static class SkillLoadoutBatchQA
    {
        const BindingFlags Hidden = BindingFlags.NonPublic | BindingFlags.Instance;
        const BindingFlags Static = BindingFlags.NonPublic | BindingFlags.Public | BindingFlags.Static;
        const string Armed = "Limitless.SkillLoadoutQA.Armed";
        static readonly List<string> results = new List<string>();
        static readonly Stack<IEnumerator> stack = new Stack<IEnumerator>();
        static string Output => Environment.GetEnvironmentVariable("LIMITLESS_B_QA_OUTPUT") ?? Path.GetFullPath(Path.Combine(Application.dataPath, "../SkillLoadoutQA"));
        static bool running;
        static bool oldOptionsEnabled;
        static EnterPlayModeOptions oldOptions;
        static SkillLoadoutBatchQA()
        {
            EditorApplication.playModeStateChanged += state =>
            {
                if (!SessionState.GetBool(Armed, false)) return;
                if (state == PlayModeStateChange.EnteredPlayMode) { running = true; stack.Push(Run()); }
                if (state == PlayModeStateChange.EnteredEditMode)
                {
                    SessionState.SetBool(Armed, false);
                    EditorSettings.enterPlayModeOptionsEnabled = oldOptionsEnabled;
                    EditorSettings.enterPlayModeOptions = oldOptions;
                    GameSaveService.FinishAudit();
                    UserSettingsService.FinishAudit();
                    Save(); EditorApplication.Exit(results.Any(line => line.StartsWith("FAIL|")) ? 1 : 0);
                }
            };
            EditorApplication.update += Tick;
        }
        public static void Launch()
        {
            string expected = "/Temp/EODIntegration20261013/Client/Assets";
            if (!Application.isBatchMode || !Application.dataPath.Replace('\\', '/').EndsWith(expected, StringComparison.OrdinalIgnoreCase))
                throw new InvalidOperationException("지정된 격리 Client의 Batch Mode에서만 실행할 수 있습니다.");
            string outputPath = Path.GetFullPath(Output);
            if (!outputPath.Replace('\\', '/').Contains("/Temp/", StringComparison.OrdinalIgnoreCase))
                throw new InvalidOperationException("QA 출력은 Temp 아래로 한정합니다.");
            EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Single);
            Directory.CreateDirectory(Output); results.Clear();
            oldOptionsEnabled = EditorSettings.enterPlayModeOptionsEnabled; oldOptions = EditorSettings.enterPlayModeOptions;
            EditorSettings.enterPlayModeOptionsEnabled = true; EditorSettings.enterPlayModeOptions = EnterPlayModeOptions.DisableDomainReload;
            GameSaveService.AuditSaveDirectory = Path.Combine(Output, "IsolatedSave");
            string savePath = Path.GetFullPath(GameSaveService.AuditSaveDirectory);
            if (!savePath.StartsWith(outputPath.TrimEnd(Path.DirectorySeparatorChar) + Path.DirectorySeparatorChar, StringComparison.OrdinalIgnoreCase))
                throw new InvalidOperationException("Save 격리 경로가 QA 출력 영역을 벗어났습니다.");
            Directory.CreateDirectory(GameSaveService.AuditSaveDirectory);
            UserSettingsService.BeginAudit(GameSaveService.AuditSaveDirectory);
            File.WriteAllText(UserSettingsService.SettingsFilePath, JsonUtility.ToJson(new UserSettingsData { MuteAll = true, SkipOpeningIntro = true }));
            Application.runInBackground = true;
            SessionState.SetBool(Armed, true); EditorApplication.EnterPlaymode();
        }
        static void Tick()
        {
            if (!running || !EditorApplication.isPlaying) return;
            try
            {
                while (stack.Count > 0)
                {
                    IEnumerator current = stack.Peek();
                    if (!current.MoveNext()) { stack.Pop(); continue; }
                    if (current.Current is IEnumerator nested) { stack.Push(nested); continue; }
                    return;
                }
                running = false; Save(); EditorApplication.ExitPlaymode();
            }
            catch (Exception e) { results.Add("FAIL|Harness|" + e); Save(); running = false; stack.Clear(); EditorApplication.ExitPlaymode(); }
        }
        static void Check(bool pass, string id, string proof)
        { results.Add((pass ? "PASS|" : "FAIL|") + id + "|" + proof); Save(); if (!pass) throw new InvalidOperationException(id); }
        static void Save() => File.WriteAllLines(Path.Combine(Output, "SkillLoadoutQA.txt"), results);
        static T Get<T>(object owner, string field) => (T)owner.GetType().GetField(field, Hidden).GetValue(owner);
        static object Call(object owner, string method, params object[] args) => owner.GetType().GetMethod(method, Hidden).Invoke(owner, args);
        static object Legacy(string method, params object[] args) => typeof(SecondRegressionAudit).GetMethod(method, Static).Invoke(null, args);
        static IEnumerator Wait(double seconds) { double end = EditorApplication.timeSinceStartup + seconds; while (EditorApplication.timeSinceStartup < end) yield return null; }
        static IEnumerator Load(string scene) => (IEnumerator)Legacy("Load", scene);
        static void Seed(string job) => Legacy("Seed", job, 1);

        static IEnumerator Run()
        {
            yield return Wait(.5);
            Seed("fighter"); yield return Load("World_StarterVillage");
            JobDefinition job = Resources.LoadAll<JobDefinition>("JobDefinitions").Single(value => value.JobId == "fighter");
            string player = PartyResourceService.PlayerCharacterId;
            var initial = SkillLoadoutService.Snapshot(player, job);
            Check(initial.Count == 3 && SkillLoadoutService.GetSlots(player, job).Count == 6, "initial", "시작3·최대6, 미습득 생성 없음");
            Check(SkillLoadoutService.TrySetSlot(player, job, 0, null) && SkillLoadoutService.Snapshot(player, job).Count == 2, "unequip", "무료 해제");
            Check(SkillLoadoutService.TrySetSlot(player, job, 5, initial[0].Id), "slot6", "실제 시작 스킬을 6번째 칸에 장착");
            Check(!SkillLoadoutService.TrySetSlot(player, job, 6, initial[0].Id) && !SkillLoadoutService.TrySetSlot(player, job, -1, initial[0].Id), "range", "7칸·음수 거절");
            Check(!SkillLoadoutService.TrySetSlot(player, job, 0, "unlearned"), "unknown", "미습득ID 거절, 정의 생성0");
            Check(SkillLoadoutService.TrySetSlot(player, job, 0, initial[0].Id) && SkillLoadoutService.GetSlots(player, job)[5] == null, "duplicate", "원슬롯 이동·중복 방지");
            Check(initial.Count == 3, "immutable", "외부 스냅샷 변경 없음");
            PlayerController.SetMovementLocked(typeof(SkillLoadoutBatchQA), true);
            Check(!SkillLoadoutService.TrySetSlot(player, job, 0, null), "cutscene-lock", "이동 잠금 중 장착 차단");
            PlayerController.SetMovementLocked(typeof(SkillLoadoutBatchQA), false);
            SkillManagementPresenter manager = SkillManagementPresenter.Instance;
            Check(manager != null, "world-manager", "월드 자동 생성"); manager.Open(); yield return null;
            Check(Get<InputAction>(manager, "toggle").bindings.Any(binding => binding.path == "<Gamepad>/start"), "gamepad-open-binding", "월드상호작용South와별도Menu/Start열기");
            Check(manager.IsOpen && PlayerController.IsMovementLocked && WorldModalState.IsOwnedBy(manager), "open-modal", "장착 UI·월드 이동 상호 배제");

            Check(slots.Count == 6 && rows.Count == 3, "management-split", "6슬롯·독립3보관함");
            slots[5].onClick.Invoke(); rows[0].onClick.Invoke();
            manager.GetComponentsInChildren<Button>().Single(value => value.name == "Equip").onClick.Invoke();
            yield return null;
            Check(SkillLoadoutService.GetSlots(player, job)[5] == initial[0].Id, "management-equip", "실제 클릭 경로6번 장착");
            foreach (var size in Sizes()) { Capture("management", size.x, size.y); yield return null; }
            // 보관함 확장 UI만 검증하는 36행 fixture입니다. 기존 정의의 행 복제이며 신규 SkillDefinition/학습 기록은 만들지 않습니다.
            RectTransform content = Get<RectTransform>(manager, "libraryContent");
            List<Button> libraryButtons = Get<List<Button>>(manager, "libraryButtons");
            List<Button> managerFocus = Get<List<Button>>(manager, "focus");
            Button template = libraryButtons[0];
            for (int i = libraryButtons.Count; i < 36; i++)
            {

            Check(libraryButtons.Last().FindSelectableOnDown() != null, "library36-navigation", "보관함 마지막행에서 패드 하단 탐색 가능");

            CompanionRosterService.UnlockSerin(); manager.Close(); manager.Open(); yield return null;

            manager.Close(); Check(!WorldModalState.IsOpen && !PlayerController.IsMovementLocked, "close-modal", "닫기 입력·이동 잠금 해제");

            yield return Load("Field_04_WesternBorder");
            Check(SkillLoadoutService.GetSlots(player, job)[5] == initial[0].Id, "scene-retain", "다른 Field 이동 시 장착 유지");

            Capture("six-cell-layout-fixture", 1280, 720);
            yield return Load("Bootstrap");
            GameSessionData.Reset(); Check(SkillLoadoutService.GetSlots(player, job)[5] == null && SkillLoadoutService.Snapshot(player, job).Count == 3, "reset", "새게임 Reset 시작3 복원");

                && report.results.All(result => result.status != "FAIL"), "skill15", "기존15스킬 실제 UI/CD/MP/취소/효과 결과15 PASS 확인");

        }
        static IEnumerable<Vector2Int> Sizes() { yield return new Vector2Int(1280, 720); yield return new Vector2Int(1600, 900); yield return new Vector2Int(1920, 1080); yield return new Vector2Int(1024, 768); yield return new Vector2Int(1680, 720); }
        // GameView 활성화 없이 실제 Canvas를 별도 Camera/RenderTexture에 그립니다.
        static void Capture(string name, int width, int height)
        {
            GameObject cameraObject = new GameObject("SkillQA_RT_Camera"); Camera camera = cameraObject.AddComponent<Camera>();
            camera.clearFlags = CameraClearFlags.SolidColor; camera.backgroundColor = new Color(.025f, .04f, .07f); camera.cullingMask = ~0;
            RenderTexture rt = new RenderTexture(width, height, 24); camera.targetTexture = rt;
            Canvas[] canvases = Object.FindObjectsByType<Canvas>(FindObjectsSortMode.None).Where(value => value.isRootCanvas && value.gameObject.activeInHierarchy).ToArray();
            foreach (Canvas canvas in canvases) { canvas.renderMode = RenderMode.ScreenSpaceCamera; canvas.worldCamera = camera; canvas.planeDistance = 1; }
            Canvas.ForceUpdateCanvases();
            var activeBattle = Object.FindAnyObjectByType<BattleSceneController>();
            if (activeBattle != null) Call(activeBattle, "PositionSkillDetailPopup");
            camera.Render();
            RenderTexture previous = RenderTexture.active; RenderTexture.active = rt;
            Texture2D texture = new Texture2D(width, height, TextureFormat.RGBA32, false); texture.ReadPixels(new Rect(0, 0, width, height), 0, 0); texture.Apply();
            File.WriteAllBytes(Path.Combine(Output, name + "_" + width + "x" + height + ".png"), texture.EncodeToPNG());
            if (name == "battle")
            {
                var controller = Object.FindAnyObjectByType<BattleSceneController>();
                Image popup = Get<Image>(controller, "skillDetailPopup"); Text message = Get<Text>(controller, "messageText");
                Check(!popup.rectTransform.Overlaps(message.rectTransform), "popup-message-gap/" + width, "상세팝업과스킬안내실패문 겹침 없음");
            }
            foreach (Canvas canvas in canvases) { canvas.renderMode = RenderMode.ScreenSpaceOverlay; canvas.worldCamera = null; }
            RenderTexture.active = previous; camera.targetTexture = null; Object.Destroy(texture); Object.Destroy(rt); Object.Destroy(cameraObject);
            Check(true, "capture/" + name + "/" + width, "실제 RenderTexture " + width + "×" + height);
        }
        private static bool Overlaps(this RectTransform first, RectTransform second)
        {
            Vector3[] a = new Vector3[4], b = new Vector3[4]; first.GetWorldCorners(a); second.GetWorldCorners(b);
            return new Rect(a[0].x, a[0].y, a[2].x - a[0].x, a[2].y - a[0].y).Overlaps(new Rect(b[0].x, b[0].y, b[2].x - b[0].x, b[2].y - b[0].y));
        }
    }
