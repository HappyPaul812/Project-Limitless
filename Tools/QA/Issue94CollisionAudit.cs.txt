 static IEnumerator Routes(string scene,string path){
  Place(new Vector2(0,-8.5f));
  var goals=scene=="Dungeon_01"?new[]{new Vector2(0,-5.1f),new Vector2(-10,0),new Vector2(10,0),new Vector2(0,8.2f),new Vector2(0,6.8f),new Vector2(0,-10.6f)}:new[]{new Vector2(0,-7.3f),new Vector2(-10,-1),new Vector2(10,-1),new Vector2(0,0),new Vector2(0,9.5f),new Vector2(0,-10.6f)};
  foreach(var goal in goals){bool ok=Walk(goal);Check(ok,scene+" "+path+" physical quest/exit route "+goal+" at="+body.position);yield return null;}
 }
 static bool Walk(Vector2 goal){
  var start=Vector2Int.RoundToInt(body.position*2);var open=new Queue<Vector2Int>();var prev=new Dictionary<Vector2Int,Vector2Int>();open.Enqueue(start);prev[start]=start;Vector2Int end=start;bool found=false;
  while(open.Count>0&&prev.Count<10000){var at=open.Dequeue();if(Vector2.Distance((Vector2)at/2,goal)<1.05f){end=at;found=true;break;}foreach(var d in new[]{Vector2Int.up,Vector2Int.down,Vector2Int.left,Vector2Int.right}){var n=at+d;var p=(Vector2)n/2;if(prev.ContainsKey(n)||Mathf.Abs(p.x)>16||Mathf.Abs(p.y)>12||!RouteClear(p))continue;prev[n]=at;open.Enqueue(n);}}
  if(!found)return false;var route=new List<Vector2>();while(end!=start){route.Add((Vector2)end/2);end=prev[end];}route.Reverse();
  foreach(var next in route){if(Vector2.Distance(body.position,goal)<.8f){Stop();return true;}int i=0;while(Vector2.Distance(body.position,next)>.09f&&i++<30)Step((next-body.position).normalized,"guided");if(i>=30){File.AppendAllText(Path.Combine(Root,"route-failures.txt"),SceneManager.GetActiveScene().name+" goal="+goal+" next="+next+" at="+body.position+" lock="+PlayerController.IsMovementLocked+" blockers="+string.Join(",",Physics2D.OverlapCircleAll(body.position,.65f).Select(c=>c.name))+"\n");Stop();return false;}}
  Stop();return Vector2.Distance(body.position,goal)<1.2f;
 }
